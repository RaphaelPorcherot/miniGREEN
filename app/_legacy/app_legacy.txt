server <- function(input, output) {
  # Affichage du titre du module
  output$module_title <- renderText({
    req(input$select_graph)
    g <- graphs_shiny[[input$select_graph]]
    mod <- input$select_graph
    n_lag <- length(which(g$x$nodes$group == "lag"))
    n_var <- ifelse(mod == "main", nrow(g$x$nodes) - n_lag, length(which(g$x$nodes$group == mod)))
    paste0("Module: ", input$select_graph, " (", n_var, " objects) ")
  })

  # Affichage du graphe
  output$graph <- renderVisNetwork({
    req(input$select_graph)
    graphs_shiny[[input$select_graph]]
  })

  # Légende (exemple basique, à adapter selon ta structure de données)
  output$legend <- renderUI({
    req(input$select_graph)
    g <- graphs_shiny[[input$select_graph]]
    nodes <- g$x$nodes
    groups <- unique(nodes$group)
    colors <- unique(nodes$color)
    # Exemple simple : affiche chaque groupe avec sa couleur
    div(
      lapply(1:length(groups), function(i) {
        tags$div(
          tags$span(
            style = paste0("display: inline-block; width: 12px; height: 12px; background-color: ", colors[i], "; margin-right: 5px;"),
            ""
          ),
          groups[i]
        )
      })
    )
  })

  # Nombre de nœuds du groupe sélectionné
  # output$node_count <- renderText({
  #   req(input$select_graph)
  #   g <- graphs_shiny[[input$select_graph]]
  #   mod <- input$select_graph
  #   length(which(g$x$nodes$group == mod))
  # })
}

shinyApp(ui = ui, server = server)
