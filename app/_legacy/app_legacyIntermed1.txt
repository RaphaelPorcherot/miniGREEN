library(bslib)
library(visNetwork)
library(dplyr)

# Charger le graphe
load("graph_obj.RData")

# Préparation
h <- 1000
w <- 1400

modules <- unique(graph_obj$nodes$type)
modules <- modules[modules != "lag"]
modules <- c("main", modules)

nodes <- graph_obj$nodes
edges <- graph_obj$edges

# Fonction utilitaire pour fixer x et y selon type et module (barycentre) — graphe principal
set_node_positions <- function(nodes_df) {
  set.seed(42) # pour reproductibilité
  
  module_levels <- unique(nodes_df$type)
  n_modules <- length(module_levels)
  
  # Barycentres répartis uniformément entre 0.1 et 0.9 sur y
  barycenters_y <- seq(0.1, 0.9, length.out = n_modules)
  names(barycenters_y) <- module_levels
  
  nodes_df <- nodes_df %>%
    rowwise() %>%
    mutate(
      x = if_else(type == "lag",
                  runif(1, 0, 0.125),        # lag sur 0-12.5% en x
                  runif(1, 0.125, 1)),       # autres sur 12.5%-100% en x
      y = barycenters_y[type] + runif(1, -0.05, 0.05) # barycentre + dispersion
    ) %>%
    ungroup()
  
  nodes_df$y <- pmin(pmax(nodes_df$y, 0), 1) # clip y entre 0 et 1
  
  nodes_df <- nodes_df %>%
    mutate(x = x * w,
           y = y * h)
  
  return(nodes_df)
}

# Fonction utilitaire pour positionner nodes du graphe focus
# selon leur degré en upstream/downstream, avec X fixé et Y libre (physique activée)
set_focus_positions <- function(nodes_df, edges_df, focus_id, up, down, w, h) {
  set.seed(42)
  
  # Calcul des degrés distance du focus
  degrees <- data.frame(id = focus_id, deg = 0)
  
  # upstream degrees
  if (up > 0) {
    current_level <- focus_id
    for (i in 1:up) {
      parents <- edges_df %>% filter(to %in% current_level) %>% pull(from) %>% unique()
      new_nodes <- setdiff(parents, degrees$id)
      if (length(new_nodes) == 0) break
      degrees <- rbind(degrees, data.frame(id = new_nodes, deg = -i))
      current_level <- new_nodes
    }
  }
  
  # downstream degrees
  if (down > 0) {
    current_level <- focus_id
    for (i in 1:down) {
      children <- edges_df %>% filter(from %in% current_level) %>% pull(to) %>% unique()
      new_nodes <- setdiff(children, degrees$id)
      if (length(new_nodes) == 0) break
      degrees <- rbind(degrees, data.frame(id = new_nodes, deg = i))
      current_level <- new_nodes
    }
  }
  
  # Merge degrés avec nodes_df
  nodes_df <- nodes_df %>%
    left_join(degrees, by = "id")
  
  # On exclut les nodes sans degré (non connectés)
  nodes_df <- nodes_df %>% filter(!is.na(deg))
  
  # Limiter deg entre -5 et 5
  nodes_df$deg <- pmin(pmax(nodes_df$deg, -5), 5)
  
  # Fonction pour position X selon degré :
  get_x_percent <- function(deg) {
    if (deg == 0) {
      return(50)
    } else if (deg < 0) {
      base <- 40 + 10 * (-deg - 1)
      return(runif(1, base, base + 10))
    } else {
      base <- 50 + 10 * (deg - 1)
      return(runif(1, base, base + 10))
    }
  }
  
  nodes_df <- nodes_df %>%
    rowwise() %>%
    mutate(
      x = get_x_percent(deg) / 100 * w,
      y = runif(1, 0.1, 0.9) * h  # y libre pour physique
    ) %>%
    ungroup()
  
  return(nodes_df)
}

# UI ----
ui <- page_navbar(
  title = "REWIND",
  theme = bs_theme(bootswatch = "flatly"),
  
  ### --- Tab 1 : Graphe principal ---
  nav_panel("REWIND",
            layout_sidebar(
              sidebar = sidebar(
                selectInput("select_graph", "Choose a graph among:", choices = modules),
                tags$hr(),
                tags$h4("Legend:"),
                uiOutput("legend")
              ),
              card(
                card_header(textOutput("module_title")),
                visNetworkOutput("graph", height = h, width = w)
              )
            )
  ),
  
  ### --- Tab 2 : Variable Focus ---
  nav_panel("Variable Focus",
            layout_sidebar(
              sidebar = sidebar(
                selectInput("focus_var", "Select variable:", choices = sort(unique(nodes$label))),
                sliderInput("focus_up", "Upstream (up):", min = 0, max = 5, value = 1),
                sliderInput("focus_down", "Downstream (down):", min = 0, max = 5, value = 1),
                tags$hr(),
                tags$h4("Legend:"),
                uiOutput("focus_legend")  
              ),
              card(
                card_header(textOutput("focus_title")),
                visNetworkOutput("focus_graph", height = h, width = w)
              )
            )
  )
)

# SERVER ----
server <- function(input, output, session) {
  
  # ---- Graph principal ----
  graph_reactive <- reactive({
    req(input$select_graph)
    
    if (input$select_graph == "main") {
      nodes_sub <- nodes
      edges_sub <- edges
    } else {
      edges_sub <- edges %>%
        filter(from_type == input$select_graph | to_type == input$select_graph)
      
      obj_in_sub <- unique(c(edges_sub$from_name, edges_sub$to_name))
      nodes_sub <- nodes %>% filter(label %in% obj_in_sub)
    }
    
    # Ajout position des nodes (barycentres fixes)
    nodes_sub <- set_node_positions(nodes_sub)
    
    # Préparation nodes avec shapes selon type
    nodes_df <- nodes_sub %>%
      mutate(
        id = id,
        label = label,
        group = type,
        color = fillcolor,
        title = label,
        x = x,
        y = y,
        shape = if_else(type == "lag", "box", "ellipse")
      ) %>% select(id, label, group, color, title, x, y, shape)
    
    edges_df <- edges_sub %>%
      rename(from = from, to = to) %>%
      select(from, to)
    
    list(nodes = nodes_df, edges = edges_df)
  })
  
  output$graph <- renderVisNetwork({
    data <- graph_reactive()
    
    visNetwork(data$nodes, data$edges, height = h, width = w) %>%
      visNodes(shape = data$nodes$shape,
               fixed = FALSE,  # Pas fixe, physique activée
               font = list(size = 20, face = "arial")) %>%
      visEdges(arrows = "to") %>%
      visOptions(
        highlightNearest = TRUE,
        nodesIdSelection = TRUE
      ) %>%
      visInteraction(
        navigationButtons = TRUE,
        zoomView = TRUE,
        dragView = TRUE
      ) %>%
      visPhysics(enabled = TRUE) %>%  # physique activée
      visEvents(
        selectNode = "function(params) {
          if(params.nodes.length > 0){
            Shiny.setInputValue('clicked_node', params.nodes[0], {priority: 'event'});
          }
        }"
      )
  })
  
  output$module_title <- renderText({
    data <- graph_reactive()
    mod <- input$select_graph
    n_lag <- sum(data$nodes$group == "lag")
    n_var <- ifelse(mod == "main",
                    nrow(data$nodes) - n_lag,
                    sum(data$nodes$group == mod))
    paste0("Module: ", mod, " (", n_var, " objects) ")
  })
  
  output$legend <- renderUI({
    data <- graph_reactive()
    nodes_df <- data$nodes
    groups <- unique(nodes_df$group)
    
    colors <- sapply(groups, function(g) {
      col <- nodes_df$color[which(nodes_df$group == g)[1]]
      if (is.na(col) | is.null(col)) "#cccccc" else col
    })
    
    div(
      lapply(seq_along(groups), function(i) {
        tags$div(
          tags$span(
            style = paste0("display: inline-block; width: 15px; height: 15px; background-color: ", colors[i], "; margin-right: 8px; border: 1px solid #666;"),
            ""
          ),
          groups[i]
        )
      })
    )
  })
  
  # ---- Graphe focus sur variable ----
  focus_graph_reactive <- reactive({
    req(input$focus_var)
    
    current_label <- input$focus_var
    current_id <- nodes$id[match(current_label, nodes$label)]
    if (is.na(current_id)) return(NULL)
    
    up <- input$focus_up
    down <- input$focus_down
    
    ids <- c(current_id)
    visited <- c()
    
    if (up > 0) {
      ids_up <- current_id
      for (i in 1:up) {
        parents <- edges %>% filter(to %in% ids_up) %>% pull(from)
        ids_up <- setdiff(parents, visited)
        visited <- c(visited, ids_up)
        ids <- c(ids, ids_up)
      }
    }
    
    if (down > 0) {
      ids_down <- current_id
      for (i in 1:down) {
        children <- edges %>% filter(from %in% ids_down) %>% pull(to)
        ids_down <- setdiff(children, visited)
        visited <- c(visited, ids_down)
        ids <- c(ids, ids_down)
      }
    }
    
    ids <- unique(ids)
    
    nodes_sub <- nodes %>% filter(id %in% ids) %>% mutate(shape = if_else(label == current_label, "box", if_else(type == "lag", "box", "ellipse")))
    edges_sub <- edges %>% filter(from %in% ids & to %in% ids)
    
    if (nrow(nodes_sub) == 0) return(NULL)
    
    # Fixer positions selon degré upstream/downstream
    nodes_sub <- set_focus_positions(nodes_sub, edges_sub, focus_id = current_id, up = up, down = down, w = w, h = h)
    
    nodes_df <- nodes_sub %>%
      mutate(
        id = id,
        label = label,
        group = type,
        color = fillcolor,
        title = label,
        x = x,
        y = y,
        shape = shape
      ) %>% select(id, label, group, color, title, x, y, shape)
    
    edges_df <- edges_sub %>%
      rename(from = from, to = to) %>%
      select(from, to)
    
    list(nodes = nodes_df, edges = edges_df)
  })
  
  output$focus_graph <- renderVisNetwork({
    data <- focus_graph_reactive()
    req(!is.null(data))
    
    visNetwork(data$nodes, data$edges, height = h, width = w) %>%
      visNodes(shape = data$nodes$shape,
               fixed = list(x = TRUE, y = FALSE), # Fixe X, Y libre pour physique
               font = list(size = 20, face = "arial")) %>%
      visEdges(arrows = "to") %>%
      visOptions(
        highlightNearest = TRUE,
        nodesIdSelection = TRUE
      ) %>%
      visInteraction(
        navigationButtons = TRUE,
        zoomView = TRUE,
        dragView = TRUE
      ) %>%
      visPhysics(enabled = TRUE) %>%  # physique activée
      visEvents(
        selectNode = "function(params) {
          if(params.nodes.length > 0){
            Shiny.setInputValue('clicked_focus_node', params.nodes[0], {priority: 'event'});
          }
        }"
      )
  })
  
  output$focus_legend <- renderUI({
    data <- focus_graph_reactive()
    req(!is.null(data))
    
    nodes_df <- data$nodes
    types <- unique(nodes_df$group)
    
    colors <- sapply(types, function(t) {
      col <- nodes_df$color[which(nodes_df$group == t)[1]]
      if (is.na(col) | is.null(col)) "#cccccc" else col
    })
    
    div(
      lapply(seq_along(types), function(i) {
        tags$div(
          tags$span(
            style = paste0("display: inline-block; width: 15px; height: 15px; background-color: ", colors[i], "; margin-right: 8px; border: 1px solid #666;"),
            ""
          ),
          types[i]
        )
      })
    )
  })
  
  output$focus_title <- renderText({
    req(input$focus_var)
    paste0("Focus on: ", input$focus_var,
           " (Up: ", input$focus_up,
           ", Down: ", input$focus_down, ")")
  })
  
  # Observe click on main graph node
  observeEvent(input$clicked_node, {
    showModal(modalDialog(
      title = "Node clicked",
      paste("You clicked node ID:", input$clicked_node),
      easyClose = TRUE
    ))
  })
  
  # Observe click on focus graph node
  observeEvent(input$clicked_focus_node, {
    showModal(modalDialog(
      title = "Focus graph node clicked",
      paste("You clicked node ID:", input$clicked_focus_node),
      easyClose = TRUE
    ))
  })
}

# Run the app
shinyApp(ui = ui, server = server)
